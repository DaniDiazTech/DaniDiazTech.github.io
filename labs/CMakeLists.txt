cmake_minimum_required(VERSION 3.20)
project(model_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
foreach(lab orientation gradient autodiff xor tensor bigram attention)
  add_executable(${lab} ${lab}.cpp)
  target_compile_options(${lab} PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME ${lab} COMMAND ${lab})
endforeach()
